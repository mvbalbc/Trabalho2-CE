\begin{frame}{Proposta de pesquisa}
\begin{itemize}
    \item \textbf{Desenhar um formulário de extração de dados} na forma de questionário para capturar variáveis do caso.
    \item \textbf{Distribuir o questionário} à população-alvo e coletar respostas.
    \item \textbf{Triar respostas} e excluir casos incompletos ou inadequados ao objetivo do estudo.
    \item \textbf{Aplicar um esquema de codificação} para transformar respostas qualitativas em variáveis quantificáveis.
    \item \textbf{Analisar estatisticamente os dados}. 
\end{itemize}
\end{frame}
