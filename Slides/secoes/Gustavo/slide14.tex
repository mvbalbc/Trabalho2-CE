\begin{frame}{Aplicação do Case Survey no Estudo}

\begin{itemize}
    \item Reúne múltiplos casos documentados e trata-os como unidades de análise comparáveis.
    \item O artigo realiza uma comparação transversal entre eles, buscando padrões gerais.
    \item O artigo converte descrições qualitativas dos casos em uma estrutura analítica quantitativa.
\end{itemize}
\end{frame}
