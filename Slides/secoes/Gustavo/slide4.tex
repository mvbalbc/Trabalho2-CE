\begin{frame}{Hipóteses}
\begin{itemize}
    \item \textbf{H1}: Maior devolução de poder ao nível local aumenta a chance de sucesso.
    \item \textbf{H2}:Inclusão de atores diversos e processos justos e inclusivos aumenta a chance de sucesso.
    \item \textbf{H3}: Apoio externo de longo prazo (como ONGs e doadores) pode aumentar o sucesso.
\end{itemize}
\end{frame}
